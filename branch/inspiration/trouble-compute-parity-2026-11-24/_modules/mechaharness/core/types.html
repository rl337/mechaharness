"""Shared domain types used across inference and harness layers.

These models form the stable public contract. Keep them serializable so
future language bindings (via FFI / IPC / OpenAPI) can share the same shapes.
"""

from __future__ import annotations

from enum import Enum
from typing import Any

from pydantic import BaseModel, Field

class Role(str, Enum):
    SYSTEM = "system"
    USER = "user"
    ASSISTANT = "assistant"
    TOOL = "tool"






[docs]
class ToolDefinition(BaseModel):
    """Provider-agnostic tool/function schema exposed to a harness."""

    name: str
    description: str = ""
    parameters: dict[str, Any] = Field(default_factory=dict)






[docs]
class ToolCall(BaseModel):
    """Model-requested tool invocation (id, name, decoded arguments)."""

    id: str
    name: str
    arguments: dict[str, Any] = Field(default_factory=dict)






[docs]
class ToolResult(BaseModel):
    """Result of executing a tool, fed back as a tool-role message."""

    tool_call_id: str
    content: str
    is_error: bool = False
    # Optional provenance/freshness (ISO-8601 or opaque). See
    # ``mechaharness.tool_result_freshness`` for staleness checks.
    source: str | None = None
    observed_at: str | None = None
    fetched_at: str | None = None
    expires_at: str | None = None






[docs]
class ChatMessage(BaseModel):
    """One turn in a chat transcript (portable across providers)."""

    role: Role
    content: str | None = None
    name: str | None = None
    tool_calls: list[ToolCall] | None = None
    tool_call_id: str | None = None
    reasoning_content: str | None = None
    extra: dict[str, Any] = Field(default_factory=dict)






[docs]
class CompletionRequest(BaseModel):
    """Normalized request handed to an InferenceStrategy."""

    model: str
    messages: list[ChatMessage]
    tools: list[ToolDefinition] | None = None
    temperature: float | None = None
    max_tokens: int | None = None
    stop: list[str] | None = None
    extra: dict[str, Any] = Field(default_factory=dict)

class Usage(BaseModel):
    """Token usage reported by a provider (fields may be unset)."""

    prompt_tokens: int | None = None
    completion_tokens: int | None = None
    total_tokens: int | None = None

    def event_payload(self) -> dict[str, int]:
        """Integer usage for ``core:inference`` payloads (missing → 0)."""
        return {
            "prompt_tokens": int(self.prompt_tokens or 0),
            "completion_tokens": int(self.completion_tokens or 0),
            "total_tokens": int(self.total_tokens or 0),

    @classmethod
    def for_event(cls, usage: Usage | None) -> dict[str, int]:
        """Normalize optional usage into an inference-event payload."""
        if usage is None:
            return cls().event_payload()
        return usage.event_payload()

class CompletionResponse(BaseModel):
    """Normalized response returned by an InferenceStrategy."""

    message: ChatMessage
    finish_reason: str | None = None
    usage: Usage | None = None
    raw: dict[str, Any] | None = None
